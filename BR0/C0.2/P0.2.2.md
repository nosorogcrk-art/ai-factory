**1. ЗАМЫСЕЛ ПАТЧА** Изоляция неопознанных файлов, обновление реестра и прямая отправка сообщения Аргусу.

**2. РЕАЛИЗАЦИЯ (L6 — КОД)**

Python

```
# Файл: quarantine.py
import os
import json
import shutil
from datetime import datetime

REGISTRY_PATH = "SYSTEM_REGISTRY.json"
QUARANTINE_DIR = "01_ЦЕХ/КАРАНТИН/"
ARGUS_MAILBOX = "01_ЦЕХ/01_ЖУРНАЛЫ/АРГУС_ВХОД/"

def send_message_to_argus(file_path, reason):
    """Отправка структурированного уведомления Аргусу (Связь BR0 -> BR1)"""
    if not os.path.exists(ARGUS_MAILBOX):
        os.makedirs(ARGUS_MAILBOX)
    
    msg_id = datetime.now().strftime("%Y%m%d_%H%M%S")
    msg = {
        "id": msg_id,
        "type": "SECURITY_ALERT",
        "subject": "File Quarantine Event",
        "body": f"File {file_path} moved to quarantine. Reason: {reason}",
        "timestamp": str(datetime.now())
    }
    with open(f"{ARGUS_MAILBOX}MSG_{msg_id}.json", "w") as f:
        json.dump(msg, f, indent=4)

def move_to_quarantine(file_path):
    """Изоляция файла и обновление реестра"""
    if not os.path.exists(QUARANTINE_DIR):
        os.makedirs(QUARANTINE_DIR)
        
    file_name = os.path.basename(file_path)
    destination = os.path.join(QUARANTINE_DIR, file_name)
    
    shutil.move(file_path, destination)
    
    # Обновление реестра
    with open(REGISTRY_PATH, 'r+') as f:
        data = json.load(f)
        data["files"][file_path] = {
            "quarantined": True,
            "reason": "not_in_registry",
            "moved_to": destination,
            "timestamp": str(datetime.now())
        }
        f.seek(0)
        json.dump(data, f, indent=4)
        f.truncate()
    
    send_message_to_argus(file_path, "not_in_registry")
```
