**1. ЗАМЫСЕЛ ПАТЧА** Реализация системы контроля файлов с поддержкой быстрого (Fast) и глубокого (Deep) сканирования.

# Файл: scanner.py
import os
import hashlib
import json
import logging

logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(message)s')
CACHE_FILE = "01_ЦЕХ/scanner_cache.json"

def get_hash(path):
    """Потоковое хэширование файла блоками по 4Кб"""
    h = hashlib.sha256()
    try:
        with open(path, "rb") as f:
            for chunk in iter(lambda: f.read(4096), b""):
                h.update(chunk)
        return h.hexdigest()
    except Exception as e:
        logging.error(f"Error hashing {path}: {e}")
        return None

def fast_scan(root="."):
    """Проверка только изменившихся файлов по mtime"""
    if not os.path.exists(CACHE_FILE):
        return deep_scan(root)
    
    with open(CACHE_FILE, 'r') as f:
        cache = json.load(f)
    
    changed = []
    for r, _, files in os.walk(root):
        for f in files:
            p = os.path.join(r, f)
            if p not in cache or os.path.getmtime(p) > cache.get(p, 0):
                changed.append(p)
    return changed

def deep_scan(root="."):
    """Полный аудит системы с пересчётом всех хэшей"""
    logging.info(f"Starting deep scan in {root}")
    inventory = {}
    new_cache = {}
    for r, _, files in os.walk(root):
        for f in files:
            p = os.path.join(r, f)
            inventory[p] = get_hash(p)
            new_cache[p] = os.path.getmtime(p)
            
    with open(CACHE_FILE, 'w') as f:
        json.dump(new_cache, f)
        
    return inventory
