# Инструкция по отправке логов в BR18 (для Cline)

1. В `main.py` добавить:
   ```python
   BR18_URL = os.getenv("BR18_URL", "http://log-aggregator:8093/api/logs")
   ENABLE_BR18 = os.getenv("ENABLE_BR18", "false").lower() == "true"
Добавить функцию:

python
async def send_log_to_br18(event_type: str, details: dict, background_tasks: BackgroundTasks) -> None:
    if not ENABLE_BR18:
        return
    async def _send():
        try:
            async with httpx.AsyncClient() as client:
                await client.post(BR18_URL, json={
                    "timestamp": datetime.now(timezone.utc).isoformat(),
                    "service": "C?.?",
                    "event_type": event_type,
                    "details": details
                }, timeout=5.0)
        except Exception as e:
            logger.error(f"Failed to send log: {e}")
    background_tasks.add_task(_send)
В эндпоинтах вызывать:

python
await send_log_to_br18("event_name", {"key": "value"}, background_tasks)
В requirements.txt добавить httpx.

В тестах мокать send_log_to_br18.
