### Назначение

Этот документ демонстрирует **эталонную реализацию патча (L6)**, соответствующую Золотому стандарту версии 5.0 и чек-листу для Cline. В качестве примера взят эндпоинт `POST /projects/{id}/messages` (добавление сообщения в проект). Пример показывает:

- Правильную структуру файлов контейнера.
- Разделение на слои (models, repositories, services, main).
- Логирование и отправку логов в BR18.
- Обработку ошибок с корректными HTTP-статусами.
- Юнит-тесты и интеграционные тесты.
- **E2E-тест с LLM-судьёй** (для демонстрации, хотя в данном патче он не обязателен, но показан как шаблон).
- Контроль цикломатической сложности и количества строк.
- Индексацию в векторную память и финальные шаги.

```python
from pydantic import BaseModel, validator

class MessageCreate(BaseModel):
    role: str
    content: str
    message_type: str = "text"

    @validator('role')
    def role_valid(cls, v):
        if v not in ('user', 'assistant'):
            raise ValueError('role must be "user" or "assistant"')
        return v

    @validator('message_type')
    def type_valid(cls, v):
        if v not in ('text', 'command', 'decision'):
            raise ValueError('message_type must be one of: text, command, decision')
        return v

class MessageResponse(BaseModel):
    id: int
    project_id: str
    role: str
    content: str
    timestamp: str
    message_type: str
```

### 2.2. repositories.py

```python
import sqlite3
from pathlib import Path

DB_PATH = Path("01_ЦЕХ/ПРОЕКТЫ/projects.db")

def insert_message(project_id: str, role: str, content: str, message_type: str, timestamp: str) -> int:
    conn = sqlite3.connect(DB_PATH)
    cur = conn.cursor()
    cur.execute(
        "INSERT INTO project_messages (project_id, role, content, timestamp, message_type) VALUES (?, ?, ?, ?, ?)",
        (project_id, role, content, timestamp, message_type)
    )
    message_id = cur.lastrowid
    conn.commit()
    conn.close()
    return message_id
```

```python
def add_message(project_id: str, role: str, content: str, message_type: str) -> dict:
    # Проверка существования проекта (упрощённо)

        raise ProjectNotFoundError(project_id)

    message_id = insert_message(project_id, role, content, message_type, timestamp)
```

### 2.5. main.py

```python
import logging
import os
from fastapi import FastAPI, HTTPException, BackgroundTasks
from .models import MessageCreate, MessageResponse
from .services import add_message
from .exceptions import ProjectNotFoundError

# ========== Логирование ==========
logger = logging.getLogger(__name__)
BR18_URL = os.getenv("BR18_URL", "http://log-aggregator:8093/api/logs")
ENABLE_BR18 = os.getenv("ENABLE_BR18", "false").lower() == "true"

async def send_log_to_br18(event_type: str, details: dict, background_tasks: BackgroundTasks) -> None:
    if not ENABLE_BR18:
        return
    import httpx
    from datetime import datetime, timezone
    async def _send():
        try:
            async with httpx.AsyncClient() as client:
                await client.post(BR18_URL, json={
                    "timestamp": datetime.now(timezone.utc).isoformat(),
                    "service": "C2.6",
                    "event_type": event_type,
                    "details": details
                }, timeout=5.0)
        except Exception as e:
            logger.error(f"Failed to send log: {e}")
    background_tasks.add_task(_send)

app = FastAPI()

@app.post("/projects/{project_id}/messages", status_code=201, response_model=MessageResponse)
async def add_message_endpoint(project_id: str, msg: MessageCreate, background_tasks: BackgroundTasks):
    try:
        result = add_message(project_id, msg.role, msg.content, msg.message_type)
        await send_log_to_br18("message_created", {"project_id": project_id, "message_id": result["id"]}, background_tasks)
        return MessageResponse(
            id=result["id"],
            project_id=project_id,
            role=msg.role,
            content=msg.content,
            timestamp=result["timestamp"],
            message_type=msg.message_type
        )
    except ProjectNotFoundError:
        await send_log_to_br18("message_create_failed", {"project_id": project_id, "error": "project_not_found"}, background_tasks)
        raise HTTPException(status_code=404, detail="Project not found")
    except Exception as e:
        logger.error(f"Unexpected error: {e}")
        await send_log_to_br18("message_create_failed", {"project_id": project_id, "error": str(e)}, background_tasks)
        raise HTTPException(status_code=500, detail="Internal server error")
```

```python
import pytest
from services import add_message
from exceptions import ProjectNotFoundError

def test_add_message_success(mocker):
    mocker.patch('services.project_exists', return_value=True)
    mocker.patch('services.insert_message', return_value=42)
    mocker.patch('services.get_current_timestamp', return_value="2026-03-24T12:00:00Z")

    assert result["id"] == 42
    assert result["timestamp"] == "2026-03-24T12:00:00Z"
```

```python
import pytest
import httpx
from main import app
from fastapi.testclient import TestClient

client = TestClient(app)

@pytest.mark.asyncio
async def test_e2e_llm_keyword_highlight():
    """
    Пример E2E-теста с LLM-судьёй для недетерминированного сценария (например, выделение ключевых слов).
    В данном патче он не обязателен, но показан как шаблон.
    """
    # 1. Вызываем реальный эндпоинт (или функцию)
    response = client.post("/projects/proj_123/messages", json={
        "role": "user", "content": "Нужна помощь с лазерной сваркой", "message_type": "text"
    })
    assert response.status_code == 201
    message_text = response.json()["content"]

    # 2. Отправляем запрос и ответ LLM-судье (например, через C19.2 или прямой вызов LLM)
    judge_prompt = {
        "prompt_id": "e2e_keyword_test",
        "input": {
            "request": "Нужна помощь с лазерной сваркой",
            "response": message_text,
            "criteria": "Ключевое слово 'лазерная сварка' должно быть выделено жирным (**...**) в ответе."
        }
    }
    # Здесь имитация вызова судьи
    judge_response = {"score": 0.95, "passed": True, "comment": "OK"}

    assert judge_response["passed"] is True
    assert judge_response["score"] >= 0.8
```

## 3. Что важно в этом примере

- **Разделение слоёв** – models, repositories, services, main.
- **Тесты** – юнит-тесты (сервис), интеграционные (API), E2E-LLM (демонстрация).
- **Обработка ошибок** – конкретные исключения, логирование, корректные статусы.
- **Логирование** – локальное (`logger.info/error`) и отправка в BR18 через `send_log_to_br18`.
- **Валидация** – через Pydantic.
- **Документация** – docstring в эндпоинтах (опущена для краткости, но должна быть).
- **Соответствие новым стандартам**:
  - Код проверен `mypy` и `ruff` (не показано, но подразумевается).
  - Цикломатическая сложность низкая (≤10).
  - Количество строк в файлах ≤300 (соблюдено).
  - E2E-тест с LLM-судьёй (шаблон).

## 5. Замечания по использованию примера

- Этот пример является эталоном **для подражания**. При реализации своих патчей следуй той же структуре и принципам.
- E2E-тесты с LLM-судьёй **обязательны** только для недетерминированных сценариев. В данном патче они показаны как иллюстрация; если твой патч не содержит таких сценариев, их можно не писать.
- Логирование в BR18 включается через переменную `ENABLE_BR18`. Пока BR18 не починен, оставляй `false`.
- Перед сдачей патча обязательно пройдись по **чек-листу для Cline** (`ЧЕК_ЛИСТ_ДЛЯ_CLINE.md`).
